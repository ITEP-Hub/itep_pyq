            // Semester 1 Toggle Function
            function toggleSemester1() {
                const content = document.getElementById('semester1Content');
                const toggleBtn = document.getElementById('semester1Toggle');
                
                // Toggle the collapsed class
                content.classList.toggle('collapsed');
                content.classList.toggle('expanded');
                
                // Change button text
                if (content.classList.contains('collapsed')) {
                    toggleBtn.textContent = '+';
                    // Save state to localStorage
                    localStorage.setItem('semester1Collapsed', 'true');
                } else {
                    toggleBtn.textContent = '-';
                    localStorage.setItem('semester1Collapsed', 'false');
                }
            }
